Driver SDK snippet

import json, socket, csv

payload = {
    "api_key": "<your instrument API key>",
    "instrument": "{{ rows[0].name if rows else 'LC-2030 #3' }}",
    "compound": "Caffeine",
    "kind": "spectrum",
    "csv": "wavelength_nm,absorbance_A\n200,0.012\n201,0.014\n...",   # or "points": [[x, y], ...]
}

# Option A — TCP (listener on {{ bridge.host or '127.0.0.1' }}:{{ bridge.port or 5056 }})
with socket.create_connection(("127.0.0.1", 5056)) as s:
    s.sendall(json.dumps(payload).encode() + b"\n")
    print(s.recv(4096).decode())
